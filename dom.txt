<script>

function showOutput(message){
    document.getElementById("output").innerText = message;
}

function demoGetElementById(){

    resetPage();

    let card = document.getElementById("student1");

    card.classList.add("highlight");

    showOutput(
        "Method: getElementById()\n\n" +
        "It selects a single element using its unique ID.\n\n" +
        "ID selected: student1"
    );
}

function demoGetElementsByClassName(){

    resetPage();

    let cards = document.getElementsByClassName("student-card");

    for(let i=0;i<cards.length;i++){
        cards[i].classList.add("highlight");
    }

    showOutput(
        "Method: getElementsByClassName()\n\n" +
        "It selects all elements having the same class name.\n\n" +
        "Class selected: student-card\n" +
        "Number of student cards selected: " + cards.length
    );
}

function demoGetElementsByTagName(){

    resetPage();

    let paragraphs = document.getElementsByTagName("p");

    for(let i=0;i<paragraphs.length;i++){
        paragraphs[i].style.color="blue";
        paragraphs[i].style.fontWeight="bold";
    }

    showOutput(
        "Method: getElementsByTagName()\n\n" +
        "It selects all elements by tag name.\n\n" +
        "Tag selected: p\n" +
        "Number of paragraph tags selected: " + paragraphs.length
    );
}

function demoQuerySelector(){

    resetPage();

    let card = document.querySelector(".student-card");

    card.classList.add("highlight");

    showOutput(
        "Method: querySelector()\n\n" +
        "It selects the first element matching a CSS selector.\n\n" +
        "Selector used: .student-card"
    );
}

function demoQuerySelectorAll(){

    resetPage();

    let cards = document.querySelectorAll(".student-card");

    cards.forEach(function(card){
        card.style.border="3px solid green";
    });

    showOutput(
        "Method: querySelectorAll()\n\n" +
        "It selects all elements matching a CSS selector.\n\n" +
        "Selector used: .student-card\n" +
        "Number of elements selected: " + cards.length
    );
}

function resetPage(){

    let cards=document.querySelectorAll(".student-card");

    cards.forEach(function(card){
        card.classList.remove("highlight");
        card.style.border="2px solid #ccc";
    });

    let paragraphs=document.querySelectorAll("p");

    paragraphs.forEach(function(p){
        p.style.color="";
        p.style.fontWeight="";
    });

    document.getElementById("output").innerText=
    "Click any button to see the result here.";
}

</script>
